# rPod additions for /boot/firmware/config.txt (docs/PLAN.md §3).
# Append this file's contents, or symlink it in, per your provisioning method.
#
# Breadboard dev board (Raspberry Pi 3B): same fragment, but keep
# dtparam=audio=on (not off), so MPD can play through the 3B's own headphone
# jack. The Zero 2 W has none: its audio is Bluetooth.

dtparam=spi=on
dtparam=i2c_arm=on
dtparam=audio=off
# The PiSugar 2's RTC, on that I2C bus beside its power chip (docs/PLAN.md
# §7.6): keeps the time while the Pi's off, or without Wi-Fi.
dtoverlay=i2c-rtc,sd3078
# Deferred with the hardware they're for (docs/PLAN.md §9, "Deferred").
# hifiberry-dac once the PCM5102A is wired: it hands the PCM peripheral to
# ALSA, so rpod-wheel then needs RPOD_WHEEL_PACING=pwm (§4.5). dwc2 for the
# USB gadget, on the Zero 2 W only: peripheral mode leaves its one USB port
# unable to take a keyboard, and the 3B has no OTG-capable port at all.
#dtoverlay=hifiberry-dac
#dtoverlay=dwc2,dr_mode=peripheral
gpu_mem=32
disable_splash=1
boot_delay=0

# Display — start with fbtft (docs/PLAN.md §5.2). Switch to the
# panel-mipi-dbi block below once the DRM path is validated.
#
# width/height are the panel's native (portrait) raster, and rotate=0 keeps
# the framebuffer that way: rpod renders 320x240 landscape and rotates it
# into the framebuffer itself (src/ui/lvgl_port.c), so frames reach the
# glass in the same row order the panel refreshes it. fbtft's own rotate=90
# wrote them across that order instead, and scrolling tore diagonally on
# every frame (docs/PLAN.md §5.3). rotate=180 is the same picture written in
# the opposite direction -- if scrolling shows a straight vertical tear on
# most frames, the panel refreshes the other way: switch to it. (rotate=90
# still works too, copied straight through, diagonal tearing and all.)
#
# Split across dtparam= lines (which apply to the dtoverlay above them) on
# purpose: the firmware silently truncates config.txt lines at 98
# characters, and as one line this cut off mid-"speed=" -- the boot log
# (`sudo vclog -m`) showed "Unknown dtparam 'spe' - ignored" and the panel
# ran at the overlay's 32 MHz default. Keep each line short.
#   speed    - SPI clock. The BCM283x divides the core clock by an even
#              number, fixed at boot from the 400 MHz core: 8, so 50 MHz,
#              and a full frame takes 25.4 ms (measured with fsync()).
#   fps      - fbtft's deferred-I/O delay is HZ/fps: 60 -> 16 ms between a
#              write and the push to the panel (default 30 -> 32 ms). rpod
#              fsync()s each frame to push it at once, so this only matters
#              to other writers.
#   txbuflen - 32 KiB SPI transfers instead of the 4 KiB (PAGE_SIZE) default,
#              so a full frame is 5 transfers, not 38.
dtoverlay=fbtft,spi0-0,st7789v,width=240,height=320,rotate=0
dtparam=reset_pin=27,dc_pin=24,led_pin=13
dtparam=speed=62500000,fps=60,txbuflen=32768
# After the fbtft lines (it edits the node they create): slows the panel's
# refresh to match that 25.4 ms push, so pushes rarely tear.
# system/overlays/rpod-panel.dts; install with `make deploy-overlay`.
dtoverlay=rpod-panel
# ...which only holds if the push stays 25.4 ms. The firmware otherwise
# drops the core clock to 275 MHz whenever the CPU is near idle -- light UI
# work like scrolling included -- and the SPI divider doesn't follow: the
# panel slowed to 34 MHz, 37.1 ms a frame, and tore on about half of them
# again. Costs a little idle power.
core_freq_min=400
